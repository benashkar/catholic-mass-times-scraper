Python Quick Start
import json, boto3, pymysql

# From Render (Secrets Manager)
client = boto3.client("secretsmanager", region_name="us-east-1")
secret = json.loads(client.get_secret_value(SecretId="/ben/ai-tool/db99")["SecretString"])
conn = pymysql.connect(host=secret["DB_HOST"], port=3306, user=secret["DB_USER"],
                       password=secret["DB_PASSWORD"], database="church_scrapes",
                       charset="utf8mb4", cursorclass=pymysql.cursors.DictCursor)

# Local dev (VPN required)
conn = pymysql.connect(host="10.10.0.8", port=3306, user="bashkar",
                       password="...", database="church_scrapes",
                       charset="utf8mb4", cursorclass=pymysql.cursors.DictCursor)
